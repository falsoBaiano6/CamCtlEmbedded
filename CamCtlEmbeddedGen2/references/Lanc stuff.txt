#define NUM_BITS 10

// Use GPT channel 0
#define GPT_CH 0

// Precompute compare times for a 10-bit frame
void prepareLancFrame(uint16_t lancByte)
{
    uint32_t t = 0;
    for (int i = 0; i < NUM_BITS; i++) {
        compareVals[i] = t;
        t += BIT_US;

    compareVals[NUM_BITS] = t;  // end boundary
}

// ─────────────────────────────────────────────────────────────
// LANC Timing
// ─────────────────────────────────────────────────────────────
const uint32_t BIT_TIME_US       = 104;
const uint32_t HALF_BIT_TIME_US  = 52;
const uint32_t SYNC_GAP_MIN_US   = 5000;   // 5 ms

// ─────────────────────────────────────────────────────────────
// Per‑Channel State
// ─────────────────────────────────────────────────────────────
struct LancChannel {
    uint8_t rxPin;
    uint8_t txPin;

    volatile uint32_t last_falling_edge;
    volatile bool     active;        // selected camera
    volatile bool     cmdPending;

    uint8_t txBuf[2];                // bytes 0–1 to inject
    uint8_t rxBuf[8];                // bytes 0–7 readback

    volatile uint8_t  byteIdx;
    volatile uint8_t  bitTick;
    volatile uint8_t  state;
};

LancChannel lanc[3];
volatile int activeChannel = -1;

// ─────────────────────────────────────────────────────────────
// State Machine States
// ─────────────────────────────────────────────────────────────
enum LancState {
    IDLE,
    SEARCHING_SYNC,
    WAITING_START,
    PROCESSING_BYTE
};

// ─────────────────────────────────────────────────────────────
// Select active channel (called by host command logic)
// ─────────────────────────────────────────────────────────────
void selectLancChannel(int idx) {
    activeChannel = idx;   // 0..2 or -1 for inactive

    for (int i = 0; i < 3; i++) {
        lanc[i].active = (i == idx);
        lanc[i].state  = SEARCHING_SYNC;
        digitalWrite(lanc[i].txPin, LOW);  // idle
    }

    lancTimer.stop();
}

// ─────────────────────────────────────────────────────────────
// Queue LANC command (bytes 0–1)
// ─────────────────────────────────────────────────────────────
void queueLancCommand(uint8_t b0, uint8_t b1) {
    if (activeChannel < 0) return;

    lanc[activeChannel].txBuf[0] = b0;
    lanc[activeChannel].txBuf[1] = b1;
    lanc[activeChannel].cmdPending = true;
}

// ─────────────────────────────────────────────────────────────
// External Falling‑Edge ISR (per channel)
// ─────────────────────────────────────────────────────────────
void lancTriggerISR(int ch) {
    LancChannel &C = lanc[ch];

    uint32_t now = micros();
    uint32_t gap = now - C.last_falling_edge;
    C.last_falling_edge = now;

    if (!C.active) return;

    switch (C.state) {

        case SEARCHING_SYNC:
            if (gap >= SYNC_GAP_MIN_US) {
                C.state   = PROCESSING_BYTE;
                C.byteIdx = 0;
                C.bitTick = 0;

                if (C.cmdPending) {
                    // Inject start bit immediately
                    digitalWrite(C.txPin, LOW);

                    lancTimer.reset();
                    lancTimer.start();
                }
            }
            break;

        case WAITING_START:
            // Start bit detected for next byte
            C.state   = PROCESSING_BYTE;
            C.bitTick = 0;

            // Inject start bit immediately
            digitalWrite(C.txPin, LOW);

            lancTimer.reset();
            lancTimer.start();
            break;

        default:
            break;
    }
}

void lancTriggerISR1() { lancTriggerISR(1); }

// ─────────────────────────────────────────────────────────────
// Timer ISR — pure bit clock
// ─────────────────────────────────────────────────────────────
void lancTimerISR(timer_callback_args_t *) {

    if (activeChannel < 0) {
        lancTimer.stop();
        return;
    }

    LancChannel &C = lanc[activeChannel];

    if (!C.cmdPending) {
        lancTimer.stop();
        return;
    }

    uint8_t tick = C.bitTick;
    uint8_t pin  = C.txPin;

void lancTimerISR(timer_callback_args_t *) {

    if (activeCam < 0) {
        lancTimer.stop();
        return;
    }

    LancChannel &C = lanc[activeCam];

    // Injection phase: bytes 0–1
    if (C.state == PROCESSING_BYTE) {

        if (!C.cmdPending) {
            lancTimer.stop();
            return;
        }

        uint8_t tick = C.bitTick;
        uint8_t pin  = C.txPin;

        // Every 2 ticks = 1 bit
        if ((tick & 1) == 1) {
            uint8_t bitIdx = tick >> 1;  // 0=start, 1..8=data, 9=stop

            if (bitIdx == 0) {
                digitalWrite(pin, LOW);   // start bit
            }
            else if (bitIdx >= 1 && bitIdx <= 8) {
                uint8_t b = C.txBuf[C.byteIdx];
                uint8_t dataBit = (b >> (bitIdx - 1)) & 1;
                digitalWrite(pin, dataBit ? HIGH : LOW);
            }
            else if (bitIdx == 9) {
                digitalWrite(pin, HIGH);  // stop bit
            }
        }

        C.bitTick++;

        // Finished 10 bits?
        if (C.bitTick >= 20) {
            C.byteIdx++;
            C.bitTick = 0;

            if (C.byteIdx >= 2) {
                // Done injecting bytes 0–1
                lancPacketComplete = true;
                C.cmdPending = false;
                C.state      = SEARCHING_SYNC;
                lancTimer.stop();
            }
            else {
                // Prepare for next byte
                C.state = WAITING_START;
                lancTimer.stop();
            }
        }
    }

    // Status read scaffolding (bytes 2–7) — not active yet, but ready
    else if (C.state == READING_STATUS) {
        // Example skeleton: sample at mid-bit and fill rxBuf[2..7]
        static uint8_t tick = 0;
        tick++;

        if ((tick & 1) == 1) {
            uint8_t bitIdx = C.statusBitIdx;  // 0=start, 1..8=data, 9=stop

            if (bitIdx >= 1 && bitIdx <= 8) {
                uint8_t bit = digitalRead(C.rxPin) ? 1 : 0;
                C.rxBuf[C.statusByteIdx] |= (bit << (bitIdx - 1));
            }

            C.statusBitIdx++;

            if (C.statusBitIdx >= 10) {
                // Finished one status byte
                C.statusBitIdx = 0;
                C.statusByteIdx++;

                if (C.statusByteIdx > 7) {
                    // Done bytes 2..7
                    C.state = SEARCHING_SYNC;
                    lancTimer.stop();
                    tick = 0;
                }
            }
        }
    }

    else {
        // Finished byte 1; command complete
        lancByteIndex = 0;
        C.cmdPending = false;
        C.state = SEARCHING_SYNC;
        lancPacketComplete = true;

        // If you want to re-arm for repeats or future commands, do it here.
        // e.g., C.cmdPending = true; for continuous zoom, etc.
    }
}
How this fixes what you saw on the scope
Byte 0:

ISR sees start bit → sets lancStartTime, lancByteIndex = 0.

Bit‑bang drives only bits 1–8.

Camera owns start, stop, and padding → waveform stays HHHLHLHH as you observed.

Byte 1:

Camera inserts padding after byte 0 stop.

Camera then drives the true start bit for byte 1.

ISR sees that falling edge → sets new lancStartTime, lancByteIndex = 1.

Bit‑bang drives only bits 1–8 of byte 1, aligned to that start.

No extra bit at the end of byte 0, no shift, no HHLHHHHH instead of HHHHHHHL.

Because zoom‑in’s direction|speed byte is 0x00, your old misalignment didn’t hurt it.
For zoom‑out (0x10), the single “1” bit was shifted into the wrong position, and the camera decoded it as something else (e.g., 0x04).

This engine removes that shift entirely.
